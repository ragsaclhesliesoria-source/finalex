Java Quiz Game

Console-Based Quiz Game - CC102

import java.util.Scanner;

public class QuizGame {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        String[] questions = {
            "1. What does CPU stand for?",
            "2. Which language styles web pages?",
            "3. What keyword creates a class in Java?",
            "4. What does HTML stand for?",
            "5. Which data type stores whole numbers?"
        };

        String[][] choices = {
            {"A. Central Processing Unit",
             "B. Computer Personal Unit",
             "C. Central Program Utility"},

            {"A. HTML", "B. CSS", "C. SQL"},

            {"A. function", "B. class", "C. define"},

            {"A. HyperText Markup Language",
             "B. HighText Machine Language",
             "C. HyperTool Main Language"},

            {"A. String", "B. double", "C. int"}
        };

        char[] answers = {'A', 'B', 'B', 'A', 'C'};

        int score = 0;

        System.out.println("==============================");
        System.out.println("       JAVA QUIZ GAME");
        System.out.println("==============================");

        for (int i = 0; i < questions.length; i++) {

            System.out.println("\n" + questions[i]);

            for (String choice : choices[i]) {
                System.out.println(choice);
            }

            System.out.print("Your answer (A/B/C): ");

            char userAnswer =
                input.next().toUpperCase().charAt(0);

            if (userAnswer == answers[i]) {
                System.out.println("Correct!");
                score++;
            } else {
                System.out.println("Incorrect!");
                System.out.println(
                    "Correct answer: " + answers[i]
                );
            }
        }

        System.out.println("\n==============================");
        System.out.println("          GAME RESULT");
        System.out.println("==============================");

        System.out.println(
            "Your score: " + score + " out of 5"
        );

        if (score == 5) {
            System.out.println("Excellent! Perfect Score!");
        } else if (score >= 3) {
            System.out.println("Good Job! Keep Learning!");
        } else {
            System.out.println("Keep Practicing!");
        }

        input.close();
    }
}
